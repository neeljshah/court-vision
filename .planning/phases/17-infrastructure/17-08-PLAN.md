    ```python
    """
    src/execution/polymarket.py — Polymarket CLOB exchange adapter.

    Architecture:
      - Market discovery: Gamma API (gamma-api.polymarket.com/markets) — NOT the CLOB API
      - Order placement: py-clob-client SDK — handles EIP-712 + Polygon nonce signing
      - Auth: L1 = EOA wallet (POLY_PRIVATE_KEY); L2 = HMAC derived via create_or_derive_api_creds()

    CRITICAL: Use Token ID (clobTokenIds[0] for YES, [1] for NO), NEVER conditionId, for orders.

    WARNING: Polymarket has NO testnet. All development must use LIVE_BETTING=0 (dry-run).
    Never flip LIVE_BETTING=1 without confirmed intent to spend real USDC on Polygon.

    Environment variables:
      POLY_PRIVATE_KEY      — Ethereum EOA private key hex (wallet holding USDC on Polygon)
      POLY_FUNDER_ADDRESS   — Wallet address (for balance checks)
    """
    import logging
    import os
    from typing import Optional

    import requests

    from .base_adapter import ExchangeAdapter, MarketQuote, OrderResult

    _log = logging.getLogger(__name__)

    _GAMMA_URL = "https://gamma-api.polymarket.com"
    _CLOB_URL = "https://clob.polymarket.com"
    _POLY_CHAIN = 137  # Polygon mainnet


    def extract_token_id(market: dict, side: str) -> str:
        """Extract the CLOB Token ID for YES or NO outcome from a Gamma API market dict.
        
        Args:
            market: market dict from Gamma API (must contain 'clobTokenIds' and 'outcomes')
            side: "yes" or "no" (case-insensitive)

        Returns:
            Token ID string for the requested outcome.

        Raises:
            ValueError: if side is not "yes" or "no"
            KeyError: if market dict is missing 'clobTokenIds'
        
        Critical: Token IDs (not conditionId) are required for order placement.
        clobTokenIds[0] = YES outcome, clobTokenIds[1] = NO outcome.
        """
        side_lower = side.lower()
        if side_lower not in ("yes", "no"):
            raise ValueError(f"side must be 'yes' or 'no', got: {side!r}")
        tokens = market["clobTokenIds"]
        return tokens[0] if side_lower == "yes" else tokens[1]


    def _win_prob_to_poly_price(win_prob: float) -> float:
        """Convert calibrated probability to Polymarket price (0.01–0.99 float)."""
        return round(max(0.01, min(0.99, win_prob)), 2)


    class PolymarketAdapter(ExchangeAdapter):
        """Polymarket CLOB adapter via py-clob-client SDK.
        
        Client is constructed lazily on first order call to avoid requiring
        POLY_PRIVATE_KEY at import time (safe for testing with mocks).
        """

        def __init__(self) -> None:
            self._client = None  # lazy init

        def _get_client(self):
            """Build and cache ClobClient with L1+L2 auth."""
            if self._client is not None:
                return self._client
            try:
                from py_clob_client.client import ClobClient
            except ImportError as exc:
                raise ImportError(
                    "py-clob-client not installed. Run: pip install py-clob-client"
                ) from exc
            private_key = os.environ.get("POLY_PRIVATE_KEY", "")
            if not private_key:
                raise EnvironmentError("POLY_PRIVATE_KEY not set")
            client = ClobClient(_CLOB_URL, key=private_key, chain_id=_POLY_CHAIN, signature_type=0)
            client.set_api_creds(client.create_or_derive_api_creds())
            self._client = client
            return client

        def find_market(self, query: str) -> list:
            """Search NBA markets on Polymarket via Gamma API.
            
            Gamma API is used for discovery; CLOB API is used for execution.
            Filters open sports markets where query appears in the question.
            """
            try:
                resp = requests.get(
                    f"{_GAMMA_URL}/markets",
                    params={"tag": "sports", "closed": "false", "limit": 100},
                    timeout=10.0,
                )
                resp.raise_for_status()
                markets = resp.json()
            except Exception as exc:
                _log.warning("Polymarket find_market failed: %s", exc)
                return []
            q = query.lower()
            return [m for m in markets if q in m.get("question", "").lower()]

        def get_quote(self, market_id: str) -> Optional[MarketQuote]:
            """Fetch best ask/bid for a market by its conditionId or slug.
            
            Returns None on error (adapter is non-fatal by design).
            yes_ask is derived from the CLOB orderbook best ask for the YES token.
            """
            try:
                resp = requests.get(
                    f"{_CLOB_URL}/book",
                    params={"token_id": market_id},
                    timeout=10.0,
                )
                resp.raise_for_status()
                book = resp.json()
            except Exception as exc:
                _log.warning("Polymarket get_quote(%s) failed: %s", market_id, exc)
                return None
            asks = book.get("asks", [])
            bids = book.get("bids", [])
            yes_ask = float(asks[0]["price"]) if asks else 0.99
            yes_bid = float(bids[0]["price"]) if bids else 0.01
            return MarketQuote(
                exchange="polymarket",
                ticker=market_id,
                yes_ask=yes_ask,
                yes_bid=yes_bid,
                available_contracts=int(float(asks[0].get("size", 0))) if asks else 0,
            )

        def place_limit_order(
            self, ticker: str, side: str, count: int, price: float
        ) -> OrderResult:
            """Place a GTC limit order on Polymarket via py-clob-client.
            
            ticker: Token ID (clobTokenIds[0] for YES, [1] for NO) — NOT conditionId.
            price: calibrated win probability (0.01–0.99); clamped before submission.
            """
            try:
                from py_clob_client.clob_types import OrderArgs, OrderType
            except ImportError as exc:
                raise ImportError("py-clob-client not installed") from exc

            poly_price = _win_prob_to_poly_price(price)
            client = self._get_client()
            order = client.create_order(OrderArgs(
                token_id=ticker,
                price=poly_price,
                size=count,
                side="BUY",
            ))
            resp = client.post_order(order, order_type=OrderType.GTC)
            order_id = resp.get("orderID", resp.get("id", "unknown"))
            _log.info("Polymarket order placed: %s @ %.2f x%d (ticker=%s)", order_id, poly_price, count, ticker)
            return OrderResult(
                exchange="polymarket",
                order_id=order_id,
                ticker=ticker,
                side=side,
                count=count,
                fill_price=poly_price,
                status=resp.get("status", "resting"),
            )
    ```
