    ```python
    """
    src/execution/kalshi.py — Kalshi exchange adapter.

    Auth: RSA-PSS signed headers (KALSHI-ACCESS-KEY, KALSHI-ACCESS-SIGNATURE, KALSHI-ACCESS-TIMESTAMP).
    Orders: Limit orders only (market orders removed 2026). yes_price is integer 1–99.
    Discovery: GET /markets?series_ticker=KXNBA&status=open — never hardcode tickers (they change per game).

    Environment variables:
      KALSHI_ACCESS_KEY       — API key from Kalshi dashboard Settings > API Keys
      KALSHI_PRIVATE_KEY_PATH — Path to RSA private key PEM file (4096-bit recommended)
      KALSHI_ENV              — "demo" (default) or "prod"

    Demo environment: https://demo-api.kalshi.co/trade-api/v2
    Prod environment: https://api.elections.kalshi.com/trade-api/v2
    """
    import base64
    import logging
    import os
    import time
    from typing import Optional

    import httpx
    from cryptography.hazmat.primitives import hashes, serialization
    from cryptography.hazmat.primitives.asymmetric import padding

    from .base_adapter import ExchangeAdapter, MarketQuote, OrderResult

    _log = logging.getLogger(__name__)

    _KALSHI_BASE = "https://api.elections.kalshi.com/trade-api/v2"
    _KALSHI_DEMO = "https://demo-api.kalshi.co/trade-api/v2"


    def _get_base_url() -> str:
        env = os.getenv("KALSHI_ENV", "demo")
        return _KALSHI_BASE if env == "prod" else _KALSHI_DEMO


    def win_prob_to_kalshi_price(win_prob: float) -> int:
        """Convert calibrated probability (0.0–1.0) to Kalshi integer cents (1–99).
        
        Kalshi valid range: 1–99 (exclusive, avoids certainty contracts).
        Clamps edge cases where LSTM may return 0.0 or 1.0.
        """
        return max(1, min(99, int(round(win_prob * 100))))


    def _load_private_key_pem() -> str:
        """Load RSA private key PEM from KALSHI_PRIVATE_KEY_PATH env var."""
        path = os.environ.get("KALSHI_PRIVATE_KEY_PATH", "")
        if not path:
            raise EnvironmentError("KALSHI_PRIVATE_KEY_PATH not set")
        with open(path, "r") as fh:
            return fh.read()


    def _make_kalshi_headers(private_key_pem: str, method: str, path: str) -> dict:
        """Generate RSA-PSS signed request headers for Kalshi Trade API v2.
        
        Args:
            private_key_pem: PEM-encoded RSA private key string
            method: HTTP method ("GET", "POST", etc.)
            path: URL path (e.g., "/trade-api/v2/portfolio/orders")
        
        Returns:
            Dict of headers including KALSHI-ACCESS-SIGNATURE.
        """
        ts = str(int(time.time() * 1000))
        msg = (ts + method.upper() + path).encode()
        key = serialization.load_pem_private_key(private_key_pem.encode(), password=None)
        sig = key.sign(
            msg,
            padding.PSS(
                mgf=padding.MGF1(hashes.SHA256()),
                salt_length=padding.PSS.DIGEST_LENGTH,
            ),
            hashes.SHA256(),
        )
        return {
            "KALSHI-ACCESS-KEY": os.environ.get("KALSHI_ACCESS_KEY", ""),
            "KALSHI-ACCESS-SIGNATURE": base64.b64encode(sig).decode(),
            "KALSHI-ACCESS-TIMESTAMP": ts,
            "Content-Type": "application/json",
        }


    class KalshiAdapter(ExchangeAdapter):
        """Kalshi REST adapter — limit orders, RSA-PSS auth, NBA KXNBA series.
        
        Set KALSHI_ENV=prod for live trading; defaults to demo environment.
        Maker orders are free (0% fee) on Kalshi as of 2026.
        """

        def __init__(self) -> None:
            self._base = _get_base_url()
            self._path_prefix = "/trade-api/v2"

        def _headers(self, method: str, path: str) -> dict:
            pem = _load_private_key_pem()
            return _make_kalshi_headers(pem, method, path)

        def find_market(self, query: str) -> list:
            """Search open KXNBA markets by player name + stat string.
            
            Always calls the API at runtime — never hardcodes tickers.
            Filters markets where query terms appear in the title (case-insensitive).
            """
            path = f"{self._path_prefix}/markets"
            params = {"status": "open", "series_ticker": "KXNBA", "limit": 200}
            try:
                resp = httpx.get(f"{self._base}/markets", params=params, timeout=10.0)
                resp.raise_for_status()
                markets = resp.json().get("markets", [])
            except Exception as exc:
                _log.warning("Kalshi find_market failed: %s", exc)
                return []
            q = query.lower()
            return [m for m in markets if q in m.get("title", "").lower()]

        def get_quote(self, market_id: str) -> Optional[MarketQuote]:
            """Fetch current best ask/bid for a market by its ticker."""
            path = f"{self._path_prefix}/markets/{market_id}"
            try:
                resp = httpx.get(f"{self._base}/markets/{market_id}", timeout=10.0)
                resp.raise_for_status()
                data = resp.json().get("market", {})
            except Exception as exc:
                _log.warning("Kalshi get_quote(%s) failed: %s", market_id, exc)
                return None
            yes_ask = data.get("yes_ask", 0) / 100.0
            yes_bid = data.get("yes_bid", 0) / 100.0
            return MarketQuote(
                exchange="kalshi",
                ticker=market_id,
                yes_ask=yes_ask,
                yes_bid=yes_bid,
                available_contracts=data.get("open_interest", 0),
            )

        def place_limit_order(
            self, ticker: str, side: str, count: int, price: float
        ) -> OrderResult:
            """Submit a limit buy order to Kalshi.
            
            price is 0.0–1.0; converted to Kalshi integer cents internally.
            side: "yes" or "no"
            """
            yes_price = win_prob_to_kalshi_price(price)
            path = f"{self._path_prefix}/portfolio/orders"
            body = {
                "ticker": ticker,
                "action": "buy",
                "side": side,
                "type": "limit",
                "count": count,
                "yes_price": yes_price,
            }
            headers = self._headers("POST", path)
            try:
                resp = httpx.post(f"{self._base}/portfolio/orders", headers=headers, json=body, timeout=15.0)
                resp.raise_for_status()
                order = resp.json().get("order", {})
            except Exception as exc:
                _log.error("Kalshi place_limit_order failed: %s", exc)
                raise
            return OrderResult(
                exchange="kalshi",
                order_id=order.get("order_id", "unknown"),
                ticker=ticker,
                side=side,
                count=count,
                fill_price=yes_price / 100.0,
                status=order.get("status", "resting"),
            )
    ```
